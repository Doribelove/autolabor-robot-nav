cmake_minimum_required(VERSION 3.10)
project(j6m_surround_capture LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(J6M_ACORE "" CACHE PATH "Matching vendor Acore source/SDK root")
find_path(VIO_INCLUDE hb_vio_interface.h HINTS "${J6M_ACORE}/hbre/camsys/libvio/include" /usr/hobot/include)
find_path(CAMERA_INCLUDE hb_camera_data_info.h HINTS "${J6M_ACORE}/hbre/camsys/libcam/include" /usr/hobot/include)
find_path(HBMEM_INCLUDE hb_mem_mgr.h HINTS "${J6M_ACORE}/hbre/libhbmem/inc" /usr/hobot/include)
find_path(GDC_INCLUDE hb_gdc_data_info.h HINTS "${J6M_ACORE}/hbre/camsys/include" /usr/hobot/include)
add_library(j6m_surround_capture SHARED capture.cpp)
target_include_directories(j6m_surround_capture PRIVATE ${VIO_INCLUDE} ${CAMERA_INCLUDE} ${HBMEM_INCLUDE})
if(GDC_INCLUDE)
  target_include_directories(j6m_surround_capture PRIVATE ${GDC_INCLUDE})
endif()
target_compile_options(j6m_surround_capture PRIVATE -Wall -Wextra -Werror)
target_link_libraries(j6m_surround_capture PRIVATE dl pthread)
install(TARGETS j6m_surround_capture LIBRARY DESTINATION lib)
option(SURROUND_BUILD_TEST_DOUBLE "Build fake vendor API for offline resource/stride tests" OFF)
if(SURROUND_BUILD_TEST_DOUBLE)
  add_library(surround_fake_vio SHARED fake_vio_test.cpp)
  target_include_directories(surround_fake_vio PRIVATE
    ${VIO_INCLUDE} ${CAMERA_INCLUDE} ${HBMEM_INCLUDE} ${GDC_INCLUDE})
  target_compile_options(surround_fake_vio PRIVATE -Wall -Wextra -Werror)
endif()
