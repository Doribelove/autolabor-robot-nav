cmake_minimum_required(VERSION 3.0.2)
project(autolabor_gps_keepout)
add_compile_options(-std=c++14)
find_package(catkin REQUIRED COMPONENTS costmap_2d geometry_msgs nav_core navfn nav_msgs pluginlib roscpp rospy std_msgs visualization_msgs message_generation)
catkin_python_setup()
add_message_files(FILES KeepoutPolygons.msg)
add_service_files(FILES KeepoutCommand.srv)
generate_messages(DEPENDENCIES geometry_msgs std_msgs)
catkin_package(INCLUDE_DIRS include LIBRARIES gps_keepout_layer gps_straight_navfn CATKIN_DEPENDS costmap_2d geometry_msgs nav_core navfn nav_msgs pluginlib roscpp rospy std_msgs visualization_msgs message_runtime)
include_directories(include ${catkin_INCLUDE_DIRS})
add_library(gps_keepout_layer src/keepout_layer.cpp)
add_dependencies(gps_keepout_layer ${${PROJECT_NAME}_EXPORTED_TARGETS} ${catkin_EXPORTED_TARGETS})
target_link_libraries(gps_keepout_layer ${catkin_LIBRARIES})
add_library(gps_straight_navfn src/straight_navfn.cpp)
target_link_libraries(gps_straight_navfn ${catkin_LIBRARIES})
catkin_install_python(PROGRAMS scripts/gps_keepout_node.py scripts/gps_reference_map_node.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(TARGETS gps_keepout_layer gps_straight_navfn LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(FILES plugins.xml global_planner_plugins.xml DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(DIRECTORY include/${PROJECT_NAME}/ DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION})
install(DIRECTORY launch config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  catkin_add_gtest(test_straight_plan test/test_straight_plan.cpp)
  if(TARGET test_straight_plan)
    target_link_libraries(test_straight_plan gps_straight_navfn ${catkin_LIBRARIES})
  endif()
  catkin_add_nosetests(test/test_geometry.py)
  catkin_add_nosetests(test/test_airport.py)
  catkin_add_nosetests(test/test_reference_map.py)
  find_package(rostest REQUIRED)
  add_rostest(test/reference_map.test)
  add_rostest_gtest(test_keepout_layer test/keepout_layer.test test/test_keepout_layer.cpp)
  add_dependencies(test_keepout_layer gps_keepout_layer ${${PROJECT_NAME}_EXPORTED_TARGETS})
  target_link_libraries(test_keepout_layer ${catkin_LIBRARIES})
endif()
