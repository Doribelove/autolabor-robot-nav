cmake_minimum_required(VERSION 3.10)
project(j6m_zed_direct_tools LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc calib3d)
find_path(OC_INCLUDE videocapture.hpp PATHS "${CAPTURE_PREFIX}/include/zed-open-capture" NO_DEFAULT_PATH REQUIRED)
find_library(OC_LIBRARY zed_open_capture PATHS "${CAPTURE_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
find_path(HID_INCLUDE hidapi.h PATHS "${CAPTURE_PREFIX}/include/hidapi" NO_DEFAULT_PATH REQUIRED)
find_library(HID_LIBRARY hidapi-libusb PATHS "${CAPTURE_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
add_executable(zed_capture_probe capture_probe.cpp)
target_compile_definitions(zed_capture_probe PRIVATE VIDEO_MOD_AVAILABLE SENSORS_MOD_AVAILABLE)
target_include_directories(zed_capture_probe PRIVATE "${OC_INCLUDE}" "${HID_INCLUDE}")
target_link_libraries(zed_capture_probe PRIVATE "${OC_LIBRARY}" "${HID_LIBRARY}" Threads::Threads)
set_target_properties(zed_capture_probe PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
add_executable(zed_opencv_check opencv_check.cpp)
target_include_directories(zed_opencv_check PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(zed_opencv_check PRIVATE ${OpenCV_LIBS})
install(TARGETS zed_capture_probe zed_opencv_check RUNTIME DESTINATION bin)
